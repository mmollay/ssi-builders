Configuration Options


                    
FilterBar Options

                    

                        
                            

                                Option
                                Type
                                Default
                                Description
                            

                        
                        
                            

                                containerId
                                string
                                -
                                Container element ID (required)
                            

                            

                                filters
                                array
                                []
                                Array of filter configurations
                            

                            

                                onChange
                                function
                                -
                                Callback when any filter changes
                            

                            

                                showRefresh
                                boolean
                                true
                                Show refresh button
                            

                            

                                showReset
                                boolean
                                true
                                Show reset button
                            

                            

                                showExport
                                boolean
                                false
                                Show export button
                            

                            

                                onRefresh
                                function
                                -
                                Refresh button handler
                            

                            

                                onExport
                                function
                                -
                                Export button handler
                            

                            

                                debounceMs
                                number
                                300
                                Debounce delay for search inputs (ms)
                            

                        
                    


                    
Filter Types

                    

                        
                            

                                Type
                                Properties
                                Description
                            

                        
                        
                            

                                select
                                key, label, options, defaultValue, includeAll
                                Dropdown select filter
                            

                            

                                search
                                key, label, placeholder, defaultValue
                                Search input with debounce
                            

                            

                                date
                                key, label, min, max, defaultValue
                                Date picker input
                            

                            

                                checkbox
                                key, label, defaultValue
                                Boolean checkbox filter
                            

                        
                    


                    
Methods

                    

                        
                            

                                Method
                                Parameters
                                Description
                            

                        
                        
                            

                                getValues()
                                -
                                Get all current filter values
                            

                            

                                getValue(key)
                                key: string
                                Get a specific filter value
                            

                            

                                setValues(values, trigger?)
                                values: object, trigger?: boolean
                                Set filter values programmatically
                            

                            

                                reset()
                                -
                                Reset all filters to defaults
                            

                            

                                updateOptions(key, options)
                                key: string, options: array
                                Update select filter options dynamically
                            

                            

                                setEnabled(key, enabled)
                                key: string, enabled: boolean
                                Enable/disable a filter
                            

                            

                                destroy()
                                -
                                Remove filter bar from DOM
