🗂️ SidebarBuilder

Collapsible navigation sidebar with search & nested items

🗂️ Navigation Sidebar

Try clicking menu items, search, and toggle collapse

Code Example:

import { SidebarBuilder } from '/vendor/ssi-builders/src/SidebarBuilder.js';

const sidebar = new SidebarBuilder({
    containerId: 'sidebarDemo',
    items: [
        {
            id: 'dashboard',
            label: 'Dashboard',
            icon: '📊',
            href: '#dashboard'
        },
        {
            id: 'products',
            label: 'Products',
            icon: '📦',
            badge: '12',
            children: [
                { id: 'all-products', label: 'All Products', href: '#products' },
                { id: 'categories', label: 'Categories', href: '#categories' },
                { id: 'add-product', label: 'Add New', href: '#add' }
            ]
        },
        {
            id: 'orders',
            label: 'Orders',
            icon: '🛒',
            badge: '5',
            href: '#orders'
        },
        {
            id: 'customers',
            label: 'Customers',
            icon: '👥',
            href: '#customers'
        },
        {
            id: 'settings',
            label: 'Settings',
            icon: '⚙️',
            children: [
                { id: 'profile', label: 'Profile', href: '#profile' },
                { id: 'security', label: 'Security', href: '#security' }
            ]
        }
    ],
    options: {
        collapsible: true,
        searchable: true,
        width: 280,
        persistent: true  // Save state to localStorage
    },
    onNavigate: (itemId) => {
        console.log('Navigated to:', itemId);
    }
});

sidebar.render();

✨ Features

  • Collapsible - Save space with collapse/expand
  • Search - Filter navigation items
  • Nested Items - Unlimited depth
  • Badges - Show counts and notifications
  • Icons - Visual navigation aids
  • Persistent State - Remembers collapse state
  • Active State - Highlights current page
  • Mobile Drawer - Responsive overlay on mobile

⚙️ Configuration Options

{
    containerId: 'sidebar',           // Container element ID
    items: [...],                     // Navigation items (required)
    options: {
        collapsible: true,            // Allow collapse/expand
        searchable: true,             // Show search box
        width: 280,                   // Sidebar width (px)
        persistent: true,             // Save state to localStorage
        defaultCollapsed: false,      // Start collapsed
        mobileBreakpoint: 768         // Mobile drawer threshold (px)
    },
    onNavigate: (itemId) => {...},   // Navigation callback
    onToggle: (collapsed) => {...}   // Collapse state callback
}
