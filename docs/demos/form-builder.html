📝 FormBuilder

Flexible forms with validation and all input types

1. Basic Form Quick Start

Simple form with text inputs, validation, and submission handling.

import { FormBuilder } from '/vendor/ssi-builders/src/index.js';

const form = new FormBuilder({
    fields: [
        {
            key: 'name',
            label: 'Full Name',
            type: 'text',
            required: true,
            placeholder: 'Enter your name'
        },
        {
            key: 'email',
            label: 'Email',
            type: 'email',
            required: true,
            placeholder: 'you@example.com'
        }
    ],
    submitLabel: 'Submit',
    onSubmit: (data) => {
        console.log('Form data:', data);
    }
});

form.render('#basic-form');

3. Validation Required, Custom Rules

Built-in validation with custom rules and error messages.

const validationForm = new FormBuilder({
    fields: [
        {
            key: 'username',
            label: 'Username',
            type: 'text',
            required: true,
            placeholder: 'Enter username',
            validate: (value) => {
                if (value.length < 3) return 'Username must be at least 3 characters';
                if (!/^[a-zA-Z0-9_]+$/.test(value)) return 'Only letters, numbers, and underscores';
                return null;
            }
        },
        {
            key: 'age',
            label: 'Age',
            type: 'number',
            required: true,
            validate: (value) => {
                if (value < 18) return 'Must be 18 or older';
                if (value > 120) return 'Invalid age';
                return null;
            }
        },
        {
            key: 'email',
            label: 'Email',
            type: 'email',
            required: true,
            placeholder: 'your@email.com'
        }
    ],
    submitLabel: 'Validate & Submit',
    onSubmit: (data) => {
        alert('Validation passed! Data: ' + JSON.stringify(data, null, 2));
    }
});

validationForm.render('#validation-form');

7. Conditional Fields Dynamic Show/Hide

Show/hide fields based on other field values.

const conditionalForm = new FormBuilder({
    fields: [
        {
            key: 'hasCompany',
            label: 'I have a company',
            type: 'checkbox'
        },
        {
            key: 'companyName',
            label: 'Company Name',
            type: 'text',
            placeholder: 'Enter company name',
            showIf: (formData) => formData.hasCompany === true
        },
        {
            key: 'vatNumber',
            label: 'VAT Number',
            type: 'text',
            placeholder: 'ATU12345678',
            showIf: (formData) => formData.hasCompany === true
        }
    ],
    submitLabel: 'Submit'
});

conditionalForm.render('#conditional-form');

10. Form State Management getValue, setValue, reset

Access and manipulate form data programmatically.

const stateForm = new FormBuilder({
    fields: [
        { key: 'title', label: 'Title', type: 'text' },
        { key: 'description', label: 'Description', type: 'textarea', rows: 3 },
        { key: 'priority', label: 'Priority', type: 'select', options: ['Low', 'Medium', 'High'] }
    ],
    submitLabel: 'Save'
});

stateForm.render('#state-form');

// Get current form data
const data = stateForm.getValue();
console.log(data); // { title: '...', description: '...', priority: '...' }

// Get specific field value
const title = stateForm.getValue('title');

// Set form data
stateForm.setValue({
    title: 'Example Title',
    description: 'Pre-filled description',
    priority: 'High'
});

// Reset form to defaults
stateForm.reset();
