📜 MenuBuilder

Dropdown & Context Menus with nested submenus

📜 Dropdown Menu

Click the button to open dropdown menu

import { MenuBuilder } from '/vendor/ssi-builders/src/MenuBuilder.js';

const menu = new MenuBuilder({
    trigger: '#dropdownBtn',
    items: [
        { id: 'new', label: 'New File', icon: '📄', shortcut: 'Ctrl+N' },
        { id: 'open', label: 'Open', icon: '📂', shortcut: 'Ctrl+O' },
        { type: 'divider' },
        { id: 'save', label: 'Save', icon: '💾', shortcut: 'Ctrl+S' },
        { id: 'saveas', label: 'Save As...', icon: '💾' },
        { type: 'divider' },
        {
            id: 'export',
            label: 'Export',
            icon: '📤',
            submenu: [
                { id: 'pdf', label: 'PDF' },
                { id: 'csv', label: 'CSV' },
                { id: 'json', label: 'JSON' }
            ]
        }
    ],
    onSelect: (itemId) => console.log('Selected:', itemId)
});

🖱️ Context Menu

Right-click on the card below

Right-click me!

Try right-clicking to see context menu

const contextMenu = new MenuBuilder({
    trigger: '#contextCard',
    triggerEvent: 'contextmenu', // Right-click
    items: [
        { id: 'copy', label: 'Copy', icon: '📋' },
        { id: 'paste', label: 'Paste', icon: '📄' },
        { type: 'divider' },
        { id: 'delete', label: 'Delete', icon: '🗑️' }
    ]
});

✨ Features

  • Dropdown & Context Menus - Click or right-click trigger
  • Nested Submenus - Unlimited depth
  • Smart Positioning - Auto-adjusts to viewport
  • Icons & Shortcuts - Visual indicators
  • Dividers - Group menu items
  • Checkable Items - Toggle states
  • Disabled Items - Conditional enable/disable
